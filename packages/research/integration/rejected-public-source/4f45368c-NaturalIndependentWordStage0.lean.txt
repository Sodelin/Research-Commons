theorem coefficient_fibres_checked :
    checkRows (project N Finset.univ)
      (GProgram.G7.SparseWordComposition.bindRows ResearchCommons.Generated.NaturalIndependentWord.Initial.certificate.rows
        (cachedStep N sample parentRegistry common encodeVariable ResearchCommons.Generated.NaturalIndependentWord.Cache.cache
          (.enter ⟨8, by decide +kernel⟩))) rows = true := by
  simp only [ResearchCommons.Generated.NaturalIndependentWord.Initial.certificate, ResearchCommons.Generated.NaturalIndependentWord.Initial.rows, GProgram.G7.SparseWordComposition.bindRows,
    List.flatMap_cons, List.flatMap_nil, List.map_cons, List.map_nil,
    ResearchCommons.Generated.NaturalIndependentWord.Stage0.cached_input0_exact,
    ResearchCommons.Generated.NaturalIndependentWord.Stage0.cached_input1_exact]
  -- Expose the existing checked payloads; only literal finite-list algebra
  -- remains before the unchanged complete output-key/coefficient check.
  simp only [ResearchCommons.Generated.NaturalIndependentWord.Step0.certificate,
    ResearchCommons.Generated.NaturalIndependentWord.Step0.entry,
    ResearchCommons.Generated.NaturalIndependentWord.Step0.rows,
    ResearchCommons.Generated.NaturalIndependentWord.Step1.certificate,
    ResearchCommons.Generated.NaturalIndependentWord.Step1.entry,
    ResearchCommons.Generated.NaturalIndependentWord.Step1.rows, GProgram.G7.EffectiveFullEpochCoefficients.product,
    List.flatMap_cons, List.flatMap_nil, List.map_cons, List.map_nil,
    List.cons_append, List.nil_append]
  -- Keep the ENTIRE original checkRows goal. Split its BOTH-side key union
  -- into literal-key obligations, avoiding one monolithic dedup/all reduction.
  -- Each obligation still checks full duplicate sums and every exponent key.
  apply checkRows_of_key_checks
  intro outputKey outputMember
  simp only [rowKeys, rows, List.map_cons, List.map_nil,
    List.mem_append, List.mem_cons, List.not_mem_nil] at outputMember
  repeat' (first | (rcases outputMember with (rfl | outputMember)) | subst outputKey | contradiction)
  all_goals
    run_tac
      Lean.Elab.Tactic.closeMainGoalUsing `kernel_decide_original_error fun target _ => do
        let target ← Lean.instantiateMVars target
        if target.hasFVar || target.hasMVar then
          throwError "closed literal coefficient goal required"
        let proof ← Lean.Meta.mkDecideProof target
        let typeLevels := (Lean.collectLevelParams {} target).params
        let lemmaLevels := (← Lean.Elab.Term.getLevelNames).reverse.filter typeLevels.contains
        let lemmaName ← Lean.withOptions (fun opts => opts.setBool `Elab.async false) do
          Lean.Meta.mkAuxLemma lemmaLevels target proof
        pure (Lean.mkConst lemmaName (lemmaLevels.map Lean.Level.param))
#check coefficient_fibres_checked
